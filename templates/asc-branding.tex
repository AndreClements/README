% asc-branding.tex — ASC visual identity for Pandoc/XeLaTeX PDF output
%
% Design system: ASC theme (andresclements.com)
% Typography: IBM Plex family (Serif body, Sans headings, Mono code)
% Palette: Earth tones — terracotta accent, burnt umber, Paynes grey
% Philosophy: paintWithScalpel — no ornamentation, let typography carry the brand
% Methodology: axiology-first aesthetics (beauty = coherent constraints)
%
% Usage: pandoc --include-in-header=templates/asc-branding.tex ...

% --- Colour palette (ASC earth tones, light-background print) ---
\usepackage{xcolor}
\definecolor{accent}{HTML}{C06A3E}          % terracotta — links, heading accents
\definecolor{burnt-umber}{HTML}{6F3A1F}     % warm secondary — citations
\definecolor{paynes-grey}{HTML}{3D4F5F}     % cool accent — code backgrounds
\definecolor{text-muted}{HTML}{666666}      % footer text, captions
\definecolor{surface}{HTML}{F5F5F0}         % off-white for code block backgrounds

% --- Typography (IBM Plex, matching ASC theme font stack) ---
% Fonts installed to user profile via tools/install-plex-fonts.ps1
% IBM Plex Mono installed system-wide (C:\Windows\Fonts)
% $asc-fonts-path$ is passed from render-pdf.ps1 (resolves per-machine)
\usepackage{fontspec}
\setmainfont{IBMPlexSerif}[
  Extension = .ttf,
  Path = {$asc-fonts-path$/},
  UprightFont = *-Regular,
  BoldFont = *-Bold,
  ItalicFont = *-Italic,
  BoldItalicFont = *-BoldItalic
]
\setsansfont{IBMPlexSans}[
  Extension = .ttf,
  Path = {$asc-fonts-path$/},
  UprightFont = *-Regular,
  BoldFont = *-Bold,
  ItalicFont = *-Italic,
  BoldItalicFont = *-BoldItalic
]
\setmonofont{IBM Plex Mono}[
  Scale = 0.9
]

% --- Emoji handled by templates/emoji-to-latex.lua (Pandoc Lua filter) ---
% Converts 🔴🟢🟡 to \textcolor{...}{$\bullet$} before LaTeX sees them

% --- Link colours ---
\usepackage[hyphens]{url}
\usepackage{hyperref}
\hypersetup{
  colorlinks = true,
  linkcolor  = accent,
  urlcolor   = accent,
  citecolor  = burnt-umber,
  filecolor  = accent,
  breaklinks = true
}

% --- Line breaking (prevent text bleeding past margins) ---
\sloppy
\emergencystretch=3em
\setlength{\hfuzz}{2pt}
% Break URLs at any character
\expandafter\def\expandafter\UrlBreaks\expandafter{\UrlBreaks\do\/\do\-\do\_\do\.\do\:\do\=\do\?\do\&\do\#}
\expandafter\def\expandafter\UrlBigBreaks\expandafter{\UrlBigBreaks\do\/\do\-}
% Allow line breaks after underscores (filenames like PROTOCOL__typed_coauthorship_relations.md)
\let\oldbfus\_
\renewcommand{\_}{\oldbfus\hspace{0pt}}

% --- Line-breaking quality (microtype + anti-orphan / anti-hyphen-stub) ---
% Stop short paragraphs from hyphenating into a stranded last-line word, and keep
% widow/orphan lines off page breaks. microtype improves line breaks throughout.
\usepackage{microtype}
\widowpenalty=10000          % no widow (paragraph's last line alone atop a page)
\clubpenalty=10000           % no orphan (paragraph's first line alone at page foot)
\brokenpenalty=10000         % don't break a page right after a hyphen
\finalhyphendemerits=100000  % avoid a hyphen on the penultimate line (it strands a fragment)
\hyphenpenalty=1500          % discourage hyphenation (moderate; \sloppy absorbs the slack)
\lefthyphenmin=3
\righthyphenmin=3

% --- Line spacing (ASC --leading-normal: 1.6) ---
\usepackage{setspace}
\setstretch{1.5}

% --- Headers and footers (fancyhdr) ---
\usepackage{fancyhdr}
\usepackage{graphicx}
\usepackage{lastpage}

\pagestyle{fancy}
\fancyhf{}  % clear defaults

% Header: no rule (paintWithScalpel — let whitespace do the work)
\renewcommand{\headrulewidth}{0pt}

% Footer: octagram mark + page number
\fancyfoot[C]{%
  \textcolor{text-muted}{%
    \raisebox{-0.15ex}{\includegraphics[height=0.8em]{$asc-templates-path$/octagram-mark-standalone}}%
    \quad\small\thepage%
  }%
}
\renewcommand{\footrulewidth}{0pt}

% Apply to plain pages (chapter/section starts) too
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{%
    \textcolor{text-muted}{%
      \raisebox{-0.15ex}{\includegraphics[height=0.8em]{$asc-templates-path$/octagram-mark-standalone}}%
      \quad\small\thepage%
    }%
  }%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% --- Title page customisation ---
\usepackage{titling}
\pretitle{%
  \begin{center}%
  \includegraphics[height=2em]{$asc-templates-path$/octagram-mark-standalone}\\[1em]%
  \LARGE\bfseries%
}
\posttitle{\end{center}\vskip 0.5em}
\preauthor{\begin{center}\large\color{text-muted}}
\postauthor{\end{center}}
\predate{\begin{center}\normalsize\color{text-muted}}
\postdate{\end{center}\vskip 2em}

% --- Section heading colours (subtle terracotta) ---
\usepackage{titlesec}
\titleformat{\section}
  {\Large\bfseries\sffamily\color{accent}}{\thesection}{1em}{}
\titleformat{\subsection}
  {\large\bfseries\sffamily}{\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\normalsize\bfseries\sffamily}{\thesubsubsection}{1em}{}

% --- Code block styling (Paynes grey tint) ---
\usepackage{tcolorbox}
\tcbuselibrary{listings,breakable}
\definecolor{codebg}{HTML}{F0F2F5}
\newtcolorbox{codeblock}{
  colback=codebg,
  colframe=paynes-grey,
  boxrule=0.4pt,
  arc=0pt,
  left=6pt,
  right=6pt,
  top=4pt,
  bottom=4pt,
  breakable
}

% Pandoc renders fenced code via Shaded+Highlighting (not the tcolorbox above);
% make those wrap long lines and inset ~2em on both sides.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{%
  commandchars=\\\{\},%
  breaklines=true, breakanywhere=true, breaksymbolleft={},%
  xleftmargin=2em, xrightmargin=2em, fontsize=\small}

% --- Blockquote styling ---
\usepackage{mdframed}
\newmdenv[
  topline=false,
  bottomline=false,
  rightline=false,
  linewidth=2pt,
  linecolor=accent,
  innerleftmargin=12pt,
  innerrightmargin=0pt,
  innertopmargin=4pt,
  innerbottommargin=4pt
]{quotebar}

% --- Table styling (clean, minimal) ---
\usepackage{booktabs}

% --- Page break after TOC ---
\let\oldtableofcontents\tableofcontents
\renewcommand{\tableofcontents}{\oldtableofcontents\clearpage}
