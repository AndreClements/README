% --- Colour palette (Credit Pilot navy-anchored, Lab-space harmonics) ---
\usepackage{xcolor}
\definecolor{cp-navy-950}{HTML}{0A2239}       % brand anchor — deepest
\definecolor{cp-navy-700}{HTML}{024777}       % primary buttons, heading accents
\definecolor{cp-navy-600}{HTML}{005A95}       % links, interactive
\definecolor{cp-navy-500}{HTML}{056EB4}       % hover states
\definecolor{cp-navy-400}{HTML}{2483CF}       % secondary interactive
\definecolor{cp-slate-700}{HTML}{55575C}      % body text
\definecolor{cp-slate-500}{HTML}{7D7F82}      % muted text, captions
\definecolor{cp-slate-100}{HTML}{E1E2E4}      % subtle backgrounds
\definecolor{cp-green}{HTML}{036527}          % READY status, success
\definecolor{cp-amber}{HTML}{794E00}          % PROCEED_WITH_GAPS, warning
\definecolor{cp-red}{HTML}{A12B35}            % BLOCKED status, red flags
\definecolor{cp-gold}{HTML}{B38E37}           % source tier: Bureau/legal
\definecolor{cp-silver}{HTML}{90939A}         % source tier: Client docs
\definecolor{cp-bronze}{HTML}{9F582D}         % source tier: LLM analysis
\definecolor{cp-copper}{HTML}{B26D55}         % source tier: Verbal/informal

% --- Link colours (navy, not terracotta) ---
\usepackage[hyphens]{url}
\usepackage{hyperref}
\hypersetup{
  colorlinks = true,
  linkcolor  = cp-navy-600,
  urlcolor   = cp-navy-600,
  citecolor  = cp-navy-700,
  filecolor  = cp-navy-600,
  breaklinks = true
}

% --- Line breaking ---
\sloppy
\emergencystretch=3em
\setlength{\hfuzz}{2pt}
\expandafter\def\expandafter\UrlBreaks\expandafter{\UrlBreaks\do\/\do\-\do\_\do\.\do\:\do\=\do\?\do\&\do\#}
\expandafter\def\expandafter\UrlBigBreaks\expandafter{\UrlBigBreaks\do\/\do\-}
\let\oldbfus\_
\renewcommand{\_}{\oldbfus\hspace{0pt}}

% --- Line spacing ---
\usepackage{setspace}
\setstretch{1.4}

% --- Headers and footers ---
\usepackage{fancyhdr}
\usepackage{graphicx}
\usepackage{lastpage}

\pagestyle{fancy}
\fancyhf{}

% Header: thin navy rule
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{cp-navy-700}\leaders\hrule height \headrulewidth\hfill}}

% Header content: icon left, document title right
\fancyhead[L]{%
  \raisebox{-0.2ex}{\includegraphics[height=1em]{$cp-logo-path$/icon-navy}}%
  \quad\textcolor{cp-slate-500}{\small\sffamily Credit Pilot}%
}
\fancyhead[R]{%
  \textcolor{cp-slate-500}{\small\sffamily\leftmark}%
}

% Footer: page number centred
\fancyfoot[C]{%
  \textcolor{cp-slate-500}{%
    \small\thepage%
  }%
}
\renewcommand{\footrulewidth}{0pt}

% Apply to plain pages too
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{%
    \textcolor{cp-slate-500}{\small\thepage}%
  }%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% --- Title page ---
\usepackage{titling}
\pretitle{%
  \begin{center}%
  \includegraphics[height=3em]{$cp-logo-path$/logo-navy}\\[1.5em]%
  \LARGE\bfseries\sffamily\color{cp-navy-950}%
}
\posttitle{\end{center}\vskip 0.5em}
\preauthor{\begin{center}\large\color{cp-slate-500}\sffamily}
\postauthor{\end{center}}
\predate{\begin{center}\normalsize\color{cp-slate-500}\sffamily}
\postdate{\end{center}\vskip 2em}

% --- Section heading colours (navy hierarchy) ---
\usepackage{titlesec}
\titleformat{\section}
  {\Large\bfseries\sffamily\color{cp-navy-700}}{\thesection}{1em}{}
\titleformat{\subsection}
  {\large\bfseries\sffamily\color{cp-navy-950}}{\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\normalsize\bfseries\sffamily}{\thesubsubsection}{1em}{}
